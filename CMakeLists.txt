cmake_minimum_required(VERSION 3.20)
project(delivery LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(DELIVERY_WITH_TORCH "Build LibTorch training and guided evaluation" ON)
if(DELIVERY_WITH_TORCH)
  find_package(Torch REQUIRED)
  # Apply LibTorch's ABI flags to ALL targets, including the shared domain API.
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
endif()
add_library(delivery_core src/domain.cpp src/geometry.cpp src/graph.cpp src/ga.cpp src/io.cpp)
target_include_directories(delivery_core PUBLIC include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(delivery_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
foreach(app generate_dataset evaluate_ga)
  add_executable(${app} apps/${app}.cpp)
  target_link_libraries(${app} PRIVATE delivery_core)
endforeach()
enable_testing()
add_executable(test_core tests/test_core.cpp)
target_link_libraries(test_core PRIVATE delivery_core)
add_test(NAME core COMMAND test_core)
if(DELIVERY_WITH_TORCH)
  add_library(delivery_gnn src/gnn.cpp)
  target_link_libraries(delivery_gnn PUBLIC delivery_core "${TORCH_LIBRARIES}")
  foreach(app train_gnn evaluate_gnn_ga)
    add_executable(${app} apps/${app}.cpp)
    target_link_libraries(${app} PRIVATE delivery_gnn)
  endforeach()
  add_executable(test_gnn tests/test_gnn.cpp)
  target_link_libraries(test_gnn PRIVATE delivery_gnn)
  add_test(NAME gnn COMMAND test_gnn)
endif()
add_executable(solve_delivery apps/solve_delivery.cpp)
if(DELIVERY_WITH_TORCH)
  target_link_libraries(solve_delivery PRIVATE delivery_gnn)
  target_compile_definitions(solve_delivery PRIVATE DELIVERY_HAS_TORCH)
else()
  target_link_libraries(solve_delivery PRIVATE delivery_core)
endif()
